{{- /* Automatic gallery for "page bundle" posts: content/posts//index.md + the photos in the same folder. Every image in the folder is shown in file-name order, auto-rotated, resized and re-encoded as WebP at build time. Re-encoding drops all EXIF metadata (GPS, camera serial, ...). The original files are not published (cascade build.publishResources = false in config.toml). Optional captions in front matter, keyed by file name: captions: 01.jpg: "Alfama at sunset" Old posts (gallery-image shortcodes, no images in the folder) are unaffected. */ -}} {{- $images := sort (.Resources.ByType "image") "Name" -}} {{- with $images -}} {{- $captions := $.Params.captions | default dict -}}
{{- range $images -}} {{- $img := . -}} {{- if ne .MediaType.SubType "svg" -}} {{- $img = .Filter images.AutoOrient -}} {{- end -}} {{- $view := $img.Process "fit 1600x1600 webp q82" -}} {{- $full := $img.Process "fit 2560x2560 webp q85" -}} {{- $caption := index $captions (lower .Name) | default "" -}} {{- end -}}
{{- end -}}
